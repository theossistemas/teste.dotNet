CREATE DATABASE Livraria;

CREATE TABLE Categoria
(
    Id UNIQUEIDENTIFIER NOT NULL PRIMARY KEY,
    Nome NVARCHAR(200) NOT NULL
);

CREATE UNIQUE INDEX IX_Categoria_Nome ON Categoria(Nome);



CREATE TABLE Livro
(
    Id UNIQUEIDENTIFIER NOT NULL PRIMARY KEY,
    Titulo NVARCHAR(300) NOT NULL,
    Autor NVARCHAR(200) NOT NULL,
    AnoPublicacao INT NOT NULL,
    ISBN NVARCHAR(20) NOT NULL,
    NumeroPaginas INT NOT NULL,
    Editora NVARCHAR(200) NOT NULL,
    CategoriaId UNIQUEIDENTIFIER NOT NULL,
    Idioma NVARCHAR(50) NOT NULL,
    Preco DECIMAL(10,2) NOT NULL,
    Sinopse NVARCHAR(MAX) NULL,
    DataCadastro DATETIME2 NOT NULL DEFAULT SYSUTCDATETIME(),
    Disponivel BIT NOT NULL DEFAULT 1,
    
    CONSTRAINT FK_Livro_Categoria FOREIGN KEY (CategoriaId) REFERENCES Categoria(Id)
);

CREATE INDEX IX_Livro_Titulo ON Livro(Titulo);
